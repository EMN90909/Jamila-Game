cmake_minimum_required(VERSION 3.20)
project(JamilaGame VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
add_library(jamila_core STATIC src/core.cpp src/mcp.cpp src/godot_agent.cpp)
target_include_directories(jamila_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(jamila_core PUBLIC PkgConfig::CURL PkgConfig::JSONC)
target_compile_options(jamila_core PRIVATE -Wall -Wextra -Wpedantic)
find_program(ZIG_EXECUTABLE zig)
if(ZIG_EXECUTABLE)
    set(ZIG_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/libjamila_zig.a")
    add_custom_command(OUTPUT "${ZIG_LIBRARY}"
        COMMAND "${ZIG_EXECUTABLE}" build-lib "${CMAKE_CURRENT_SOURCE_DIR}/src/zig/digest.zig" -O ReleaseSafe -femit-bin="${ZIG_LIBRARY}"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/zig/digest.zig" VERBATIM)
    add_custom_target(jamila_zig_build DEPENDS "${ZIG_LIBRARY}")
    add_library(jamila_zig STATIC IMPORTED GLOBAL)
    set_target_properties(jamila_zig PROPERTIES IMPORTED_LOCATION "${ZIG_LIBRARY}")
    add_dependencies(jamila_zig jamila_zig_build)
    target_link_libraries(jamila_core PUBLIC jamila_zig)
    target_compile_definitions(jamila_core PRIVATE JAMILA_WITH_ZIG=1)
endif()
add_executable(jamila-game src/main.cpp)
target_link_libraries(jamila-game PRIVATE jamila_core)
find_package(X11)
if(X11_FOUND)
    target_sources(jamila-game PRIVATE src/gui_x11.cpp)
    target_include_directories(jamila-game PRIVATE ${X11_INCLUDE_DIR})
    target_link_libraries(jamila-game PRIVATE ${X11_LIBRARIES})
    target_compile_definitions(jamila-game PRIVATE JAMILA_WITH_X11=1)
endif()
if(BUILD_TESTING)
    add_executable(core_test tests/core_test.cpp)
    target_link_libraries(core_test PRIVATE jamila_core)
    target_compile_options(core_test PRIVATE -UNDEBUG)
    add_executable(mock_godot_mcp tests/mock_mcp.cpp)
    target_link_libraries(mock_godot_mcp PRIVATE PkgConfig::JSONC)
    target_compile_options(mock_godot_mcp PRIVATE -UNDEBUG)
    add_executable(mcp_agent_test tests/mcp_agent_test.cpp)
    target_link_libraries(mcp_agent_test PRIVATE jamila_core)
    target_compile_options(mcp_agent_test PRIVATE -UNDEBUG)
    add_test(NAME mcp_agent COMMAND mcp_agent_test $<TARGET_FILE:mock_godot_mcp>)
    set_tests_properties(mcp_agent PROPERTIES TIMEOUT 30)
    add_test(NAME core COMMAND core_test)
    add_test(NAME cli_help COMMAND jamila-game --help)
endif()
